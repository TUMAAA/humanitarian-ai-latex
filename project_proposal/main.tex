% Do not \input{titlepage} here: the root file's preamble already loads
% it (via \input{\subproject/titlepage}), and subfiles copies that same
% preamble in when this file is compiled standalone, so \proposaltitle
% and \proposalauthor are already defined by the time we get here. A
% second \input would redefine them with \newcommand and error out.
% Consequence: compiling this file standalone uses whichever subproject's
% titlepage root/main.tex currently has \subproject set to -- set that to
% project_proposal first if you want this subproject's own title/author.

\begin{document}
\title{\proposaltitle}
\author{\proposalauthor}
\maketitle

\input{citation_macros}
\input{introduction}
